// ============================================================================
//  tampermonkey.txt — userscript для получения -pipe / -token (TunVPN)
//  Удобная альтернатива cred.txt: ставится один раз, дальше всё автоматически.
//
//  УСТАНОВКА:
//   1. Поставь расширение Tampermonkey (Chrome/Firefox/Edge): https://www.tampermonkey.net
//   2. Значок Tampermonkey → "Create a new script…" (Создать новый скрипт).
//   3. Удали содержимое редактора, вставь ВЕСЬ этот файл, сохрани (Ctrl+S).
//
//  ИСПОЛЬЗОВАНИЕ:
//   4. Открой https://notes.mail.ru (залогиненным). Справа сверху появится
//      бейдж "grabber активен · жду заметки".
//   5. Открой любую заметку. Как только найдутся pipe и token, на весь экран
//      покажется токен с кнопками: "КОПИРОВАТЬ ТОКЕН" и "pipe+token".
//   6. Жми "pipe+token" — в буфер попадёт готовая строка "-pipe … -token …".
//      Вставь pipe и token в поля приложения TunVPN (или в запуск ./tun -client).
//
//  Команды в консоли (DevTools): wsShow() wsCreds() wsCopy() wsList() wsHex(i)
//
//  ВАЖНО: pipe/token — твои личные креды сессии; не делись. Пара протухает —
//  если туннель отвечает "Unauthorized (4401)", открой заметку заново и возьми
//  свежую пару. Скрипт только читает трафик твоей вкладки, никуда не отправляет.
// ============================================================================

// ==UserScript==
// @name         notes.mail.ru — pipe/token grabber
// @namespace    local.ws.sniffer
// @version      6.0
// @description  Перехват WebSocket notes.mail.ru, извлечение pipe/token, полноэкранный вывод токена
// @match        *://*.mail.ru/*
// @match        *://*/*
// @run-at       document-start
// @all-frames   true
// @grant        unsafeWindow
// ==/UserScript==

(function () {
  'use strict';

  const URL_FILTER = /notes\.mail\.ru/i;   // null = все сокеты

  // Работаем в реальном окне страницы (обход изолированного мира и CSP)
  const W = (typeof unsafeWindow !== 'undefined' && unsafeWindow) ? unsafeWindow : window;

  const NativeWS = W.WebSocket;
  if (!NativeWS) { console.warn('[grabber] нет WebSocket в этом окне'); return; }
  if (NativeWS.__grabbed) return;

  const L = W.__ws = [];
  const enc = new TextEncoder();
  const t = () => new Date().toISOString().substr(11, 12);
  const asStr = u8 => new TextDecoder('utf-8', { fatal: false }).decode(u8);
  const hex = (u8, n) => Array.from(u8.slice(0, n || u8.length)).map(b => b.toString(16).padStart(2, '0')).join(' ');
  const ascii = (u8, n) => Array.from(u8.slice(0, n || u8.length)).map(b => (b >= 32 && b < 127) ? String.fromCharCode(b) : '.').join('');

  const creds = W.__creds = { pipeid: null, token: null, __done: false };
  const RE_TOKEN = /o:[0-9a-f]+\.c:\d+/i;
  const RE_UUID = /[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}/i;
  const credLine = () => "-pipe '$" + creds.pipeid + "'  -token '" + creds.token + "'";

  function scan(bytes, url) {
    const s = asStr(bytes);
    if (!creds.token) { const m = s.match(RE_TOKEN); if (m) { creds.token = m[0]; badge('token найден', '#f1c40f'); console.log('%c[+] token =', 'color:#f1c40f', m[0]); } }
    if (!creds.pipeid) { const m = (url || '').match(RE_UUID) || s.match(RE_UUID); if (m) { creds.pipeid = m[0]; console.log('%c[+] pipeid =', 'color:#f1c40f', m[0]); } }
    if (creds.pipeid && creds.token && !creds.__done) {
      creds.__done = true;
      console.log('%c' + credLine(), 'color:#2ecc71;font-weight:bold');
      showOverlay();
    }
  }

  function push(dir, bytes, url) {
    const e = { time: t(), ts: Date.now(), dir, url, bytes, len: bytes.length };
    L.push(e); if (L.length > 20000) L.shift();
    const c = dir === '>>' ? '#e67e22' : '#27ae60';
    console.log('%c' + dir + ' ' + e.time + ' [' + e.len + 'b]', 'color:' + c + ';font-weight:bold', hex(bytes, 20) + (e.len > 20 ? ' …' : ''), '|', ascii(bytes, 40));
    scan(bytes, url);
  }
  function toBytes(d, cb) {
    if (typeof d === 'string') return cb(enc.encode(d));
    if (d instanceof W.ArrayBuffer || d instanceof ArrayBuffer) return cb(new Uint8Array(d.slice(0)));
    if (ArrayBuffer.isView(d)) return cb(new Uint8Array(d.buffer.slice(d.byteOffset, d.byteOffset + d.byteLength)));
    if (d && typeof d.arrayBuffer === 'function') return void d.arrayBuffer().then(b => cb(new Uint8Array(b)));
    cb(enc.encode(String(d)));
  }
  const watched = u => !URL_FILTER || URL_FILTER.test(String(u));

  // ---------- хук: перехват send на прототипе ----------
  const origSend = NativeWS.prototype.send;
  NativeWS.prototype.send = function (d) {
    try { if (this.__watch) toBytes(d, b => push('>>', b, this.__url)); } catch (e) { }
    return origSend.apply(this, arguments);
  };

  // ---------- хук: обёртка конструктора ----------
  function WSWrap(url, protocols) {
    const ws = protocols !== undefined ? new NativeWS(url, protocols) : new NativeWS(url);
    ws.__url = String(url);
    ws.__watch = watched(url);
    if (ws.__watch) {
      console.log('%c-- CONNECT ' + ws.__url, 'color:#4da3ff');
      const mu = ws.__url.match(RE_UUID);
      if (mu && !creds.pipeid) { creds.pipeid = mu[0]; console.log('%c[+] pipeid (url) =', 'color:#f1c40f', mu[0]); }
      ws.addEventListener('message', ev => { try { toBytes(ev.data, b => push('<<', b, ws.__url)); } catch (e) { } });
    }
    return ws;
  }
  WSWrap.prototype = NativeWS.prototype;
  ['CONNECTING', 'OPEN', 'CLOSING', 'CLOSED'].forEach((k, i) => { WSWrap[k] = i; });
  WSWrap.__grabbed = true;
  try { Object.defineProperty(WSWrap, 'name', { value: 'WebSocket' }); } catch (e) { }
  WSWrap.toString = () => 'function WebSocket() { [native code] }';
  try { W.WebSocket = WSWrap; } catch (e) { console.warn('[grabber] не удалось заменить WebSocket', e); }

  // ================= UI =================
  function copyText(str, btn) {
    const done = () => { if (btn) { const o = btn.textContent; btn.textContent = 'СКОПИРОВАНО ✓'; setTimeout(() => btn.textContent = o, 1500); } };
    if (navigator.clipboard) navigator.clipboard.writeText(str).then(done, () => fb());
    else fb();
    function fb() { const ta = document.createElement('textarea'); ta.value = str; ta.style.cssText = 'position:fixed;opacity:0'; document.body.appendChild(ta); ta.select(); try { document.execCommand('copy'); done(); } catch (e) { } ta.remove(); }
  }

  function badge(text, color) {
    if (!document.body) { document.addEventListener('DOMContentLoaded', () => badge(text, color)); return; }
    let b = document.getElementById('__wsBadge');
    if (!b) {
      b = document.createElement('div'); b.id = '__wsBadge';
      b.style.cssText = 'position:fixed;top:10px;right:10px;z-index:2147483647;padding:8px 14px;border-radius:8px;' +
        'font:600 13px/1 Menlo,Consolas,monospace;color:#fff;box-shadow:0 2px 10px rgba(0,0,0,.4)';
      document.body.appendChild(b);
    }
    b.style.background = color || '#4da3ff';
    b.textContent = text;
  }

  function showOverlay() {
    if (!document.body) { document.addEventListener('DOMContentLoaded', showOverlay); return; }
    const old = document.getElementById('__wsOverlay'); if (old) old.remove();
    const ov = document.createElement('div');
    ov.id = '__wsOverlay';
    ov.style.cssText = 'position:fixed;inset:0;z-index:2147483647;background:rgba(10,12,16,.97);display:flex;' +
      'flex-direction:column;align-items:center;justify-content:center;gap:22px;font-family:Menlo,Consolas,monospace;padding:5vw;box-sizing:border-box';
    const bs = bg => 'padding:14px 26px;font:600 17px/1 inherit;color:#fff;background:' + bg + ';border:0;border-radius:10px;cursor:pointer';
    ov.innerHTML =
      '<div style="color:#8b98a5;font-size:14px;letter-spacing:1px">TOKEN</div>' +
      '<div style="color:#2ecc71;font-size:clamp(16px,3.2vw,34px);font-weight:700;text-align:center;word-break:break-all;line-height:1.35;max-width:92vw">' + creds.token + '</div>' +
      '<div style="color:#e67e22;font-size:clamp(13px,2vw,20px);text-align:center;word-break:break-all;max-width:92vw"><span style="color:#8b98a5">pipe:</span> $' + creds.pipeid + '</div>' +
      '<div style="display:flex;gap:12px;flex-wrap:wrap;justify-content:center;margin-top:8px">' +
      '<button id="__cT" style="' + bs('#27ae60') + '">КОПИРОВАТЬ ТОКЕН</button>' +
      '<button id="__cA" style="' + bs('#2980b9') + '">pipe+token</button>' +
      '<button id="__cX" style="' + bs('#555') + '">ЗАКРЫТЬ (Esc)</button></div>';
    document.body.appendChild(ov);
    ov.querySelector('#__cT').onclick = function () { copyText(creds.token, this); };
    ov.querySelector('#__cA').onclick = function () { copyText(credLine(), this); };
    const close = () => { ov.remove(); document.removeEventListener('keydown', onKey); };
    ov.querySelector('#__cX').onclick = close;
    const onKey = e => { if (e.key === 'Escape') close(); };
    document.addEventListener('keydown', onKey);
    ov.addEventListener('click', e => { if (e.target === ov) close(); });
  }

  // ---------- команды ----------
  const g = (k, f) => { try { W[k] = f; } catch (e) { } window[k] = f; };
  g('wsShow', showOverlay);
  g('wsCreds', () => { console.log('pipeid:', creds.pipeid, '\ntoken :', creds.token); return creds; });
  g('wsCopy', () => { copyText(credLine()); return credLine(); });
  g('wsList', () => console.table(L.map((e, i) => ({ i, time: e.time, dir: e.dir, len: e.len, head: hex(e.bytes, 12) }))));
  g('wsHex', (i, max) => { const u = L[i].bytes, n = Math.min(u.length, max || 512); let s = ''; for (let o = 0; o < n; o += 16) { const r = u.subarray(o, o + 16); s += o.toString(16).padStart(6, '0') + '  ' + hex(r).padEnd(47) + '  ' + ascii(r) + '\n'; } console.log(s); });

  // бейдж загрузки — виден сразу, подтверждает, что скрипт инжектнулся
  badge('grabber активен · жду заметки', '#4da3ff');
  console.log('%c[notes.mail.ru grabber v6] активен на ' + location.host + '. unsafeWindow=' + (W !== window) + '. Зайдите в заметки.', 'color:#4da3ff;font-weight:bold');
})();
